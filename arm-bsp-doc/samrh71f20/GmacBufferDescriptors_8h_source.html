154static inline Gmac_RxBufferDescriptorEntry Gmac_rxBufferDescriptorEncodeEntry(
155 const Gmac_RxBufferParameters* const bufferParameters)
156{
157 const Gmac_RxBufferDescriptorEntry entry = {
158 // cppcheck-suppress misra-c2012-11.6
159 .descl
160 = BIT_FIELD_VALUE(GMAC_RXBUF_DESCL_START_ADDRESS, ((uint32_t)bufferParameters->address) >> 2u)
161 | BIT_VALUE(GMAC_RXBUF_DESCL_WRAP, bufferParameters->isLastDescriptorInQueue),
162 .desch = 0u,
163 };
164
165 return entry;
166}

379{

381 // cppcheck-suppress misra-c2012-11.6

387 };
