156static inline bool swapBitAtOffset(volatile uint32_t* const reg,
157 const uint32_t offset,
158 const bool value)
159{
160 const uint32_t current = *reg;
161 if (value)
162 {
163 *reg = current | (UINT32_C(1) << offset);
164 }
165 else
166 {
167 *reg = current & ~(UINT32_C(1) << offset);
168 }
169
170 return isBitSet(current, offset);
171}
